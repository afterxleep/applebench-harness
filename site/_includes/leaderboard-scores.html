{%- comment -%} The ranked AppleBench score bars for the current suite revision — the leaderboard and nothing else. Models still working through the current suite are left out entirely rather than shown as pending; the full comparison, including them, lives on the results page. Params: runs (site.benchmarks). {%- endcomment -%} {%- assign current = site.data.suite_revisions | where: "current", true | first -%} {%- assign by_score = "" | split: "" -%} {%- for run in include.runs -%} {%- assign rows = site.data.benchmarks[run.data] -%} {%- assign report = site.data.reports[run.data] -%} {%- if rows and report and current and run.suite_revision == current.id and rows.size == current.gold_tasks -%} {%- assign score_key = 0 -%} {%- if report.score.available > 0 %} {%- assign score_key = report.score.points | times: 10000 | divided_by: report.score.available | round %} {%- endif -%} {%- assign name = run.model | split: "/" | last -%} {%- assign effort_label = "(default)" -%} {%- for reasoning in report.reasoning -%} {%- if reasoning.effort -%} {%- assign effort_label = reasoning.effort | prepend: "(" | append: ")" -%} {%- break -%} {%- endif -%} {%- endfor -%} {%- capture score_row -%}{{ score_key | prepend: "00000" | slice: -5, 5 }}|{{ name }}|{{ effort_label }}|{{ report.score.points }}|{{ report.score.available }}|{{ run.url }}{%- endcapture -%} {%- assign by_score = by_score | push: score_row -%} {%- endif -%} {%- endfor -%} {%- assign by_score = by_score | sort | reverse -%}
